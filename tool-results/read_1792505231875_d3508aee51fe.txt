    62→function isOverdue(job: Delivery): boolean {
    63→  if (!job.expectedAt) return false;
    64→  if (job.status === "DELIVERED" || job.status === "FAILED" || job.status === "CANCELLED") return false;
    65→  return new Date(job.expectedAt).getTime() < Date.now();
    66→}

   168→// ─────────────────────────────────────────────────────────────
   169→// Signature pad — pointer events (touch + mouse + stylus),
   170→// devicePixelRatio-scaled canvas, strokes stored for crisp redraws.
   171→// ─────────────────────────────────────────────────────────────
   172→
   173→function SignaturePad({
   174→  canvasRef,
   175→  onInkChange,
   176→}: {
   177→  canvasRef: RefObject<HTMLCanvasElement | null>;
   178→  onInkChange: (hasInk: boolean) => void;
   179→}) {
   180→  const strokesRef = useRef<Array<Array<{ x: number; y: number }>>>([]);
   181→  const drawingRef = useRef(false);
   182→  const [inked, setInked] = useState(false);
   183→
   184→  const applyStyle = useCallback((ctx: CanvasRenderingContext2D) => {
   185→    ctx.lineWidth = 2.5;
   186→    ctx.strokeStyle = "#1e293b";
   187→    ctx.fillStyle = "#1e293b";
   188→    ctx.lineCap = "round";
   189→    ctx.lineJoin = "round";
   190→  }, []);
   191→
   192→  const redraw = useCallback(() => {
   193→    const canvas = canvasRef.current;
   194→    const ctx = canvas?.getContext("2d");
   195→    if (!canvas || !ctx) return;
   196→    const rect = canvas.getBoundingClientRect();
   197→    applyStyle(ctx);
   198→    ctx.clearRect(0, 0, rect.width, rect.height);
   199→    for (const stroke of strokesRef.current) {
   200→      if (stroke.length === 0) continue;
   201→      ctx.beginPath();
   202→      ctx.moveTo(stroke[0].x, stroke[0].y);
   203→      if (stroke.length === 1) {
   204→        ctx.arc(stroke[0].x, stroke[0].y, 1.25, 0, Math.PI * 2);
   205→        ctx.fill();
   206→        continue;
   207→      }
   208→      for (let i = 1; i < stroke.length; i++) ctx.lineTo(stroke[i].x, stroke[i].y);
   209→      ctx.stroke();
   210→    }
   211→  }, [applyStyle, canvasRef]);
   212→
   213→  // Scale the backing store by devicePixelRatio for crisp lines; re-run on resize
   214→  // (dialog open animation, rotation) and replay stored strokes.
   215→  useEffect(() => {
   216→    const canvas = canvasRef.current;
   217→    if (!canvas) return;
   218→    const resize = () => {
   219→      const rect = canvas.getBoundingClientRect();
   220→      if (rect.width < 2 || rect.height < 2) return;
   221→      const dpr = window.devicePixelRatio || 1;
   222→      canvas.width = Math.round(rect.width * dpr);
   223→      canvas.height = Math.round(rect.height * dpr);
   224→      const ctx = canvas.getContext("2d");
   225→      if (!ctx) return;
   226→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   227→      redraw();
   228→    };
   229→    resize();
   230→    const observer = new ResizeObserver(resize);
   231→    observer.observe(canvas);
   232→    return () => observer.disconnect();
   233→  }, [canvasRef, redraw]);
   234→
   235→  const pointerPos = (e: ReactPointerEvent<HTMLCanvasElement>) => {
   236→    const rect = e.currentTarget.getBoundingClientRect();
   237→    return { x: e.clientX - rect.left, y: e.clientY - rect.top };
   238→  };
   239→
   240→  const handlePointerDown = (e: ReactPointerEvent<HTMLCanvasElement>) => {
   241→    e.preventDefault();
   242→    try {
   243→      e.currentTarget.setPointerCapture(e.pointerId);
   244→    } catch {
   245→      /* pointer already released */
   246→    }
   247→    drawingRef.current = true;
   248→    const ctx = e.currentTarget.getContext("2d");
   249→    if (!ctx) return;
   250→    applyStyle(ctx);
   251→    const p = pointerPos(e);
   252→    strokesRef.current.push([p]);
   253→    ctx.beginPath();
   254→    ctx.arc(p.x, p.y, 1.25, 0, Math.PI * 2);
   255→    ctx.fill();
   256→  };
   257→
   258→  const handlePointerMove = (e: ReactPointerEvent<HTMLCanvasElement>) => {
   259→    if (!drawingRef.current) return;
   260→    const stroke = strokesRef.current[strokesRef.current.length - 1];
   261→    const ctx = e.currentTarget.getContext("2d");
   262→    if (!stroke || !ctx) return;
   263→    const p = pointerPos(e);
   264→    const prev = stroke[stroke.length - 1];
   265→    applyStyle(ctx);
   266→    ctx.beginPath();
   267→    ctx.moveTo(prev.x, prev.y);
   268→    ctx.lineTo(p.x, p.y);
   269→    ctx.stroke();
   270→    stroke.push(p);
   271→    if (stroke.length >= 2 && !inked) {
   272→      setInked(true);
   273→      onInkChange(true);
   274→    }
   275→  };
   276→
   277→  const endStroke = () => {
   278→    drawingRef.current = false;
   279→  };
   280→
   281→  const clear = () => {
   282→    strokesRef.current = [];
   283→    drawingRef.current = false;
   284→    const canvas = canvasRef.current;
   285→    const ctx = canvas?.getContext("2d");
   286→    if (canvas && ctx) {
   287→      const rect = canvas.getBoundingClientRect();
   288→      applyStyle(ctx);
   289→      ctx.clearRect(0, 0, rect.width, rect.height);
   290→    }
   291→    setInked(false);
   292→    onInkChange(false);
   293→  };
   294→
   295→  return (
   296→    <div className="flex flex-col gap-2">
   297→      <div className="flex items-center justify-between gap-2">
   298→        <Label>Recipient signature</Label>
   299→        <Button type="button" variant="ghost" size="sm" className="h-9 gap-1.5" onClick={clear}>
   300→          <Eraser className="size-3.5" aria-hidden="true" /> Clear
   301→        </Button>
   302→      </div>
   303→      <canvas
   304→        ref={canvasRef}
   305→        onPointerDown={handlePointerDown}
   306→        onPointerMove={handlePointerMove}
   307→        onPointerUp={endStroke}
   308→        onPointerCancel={endStroke}
   309→        onLostPointerCapture={endStroke}
   310→        className="h-44 w-full touch-none cursor-crosshair rounded-lg border bg-white"
   311→        role="img"
   312→        aria-label="Signature pad — draw the recipient's signature with your finger or a stylus"
   313→      />
   314→      <p className="text-xs text-muted-foreground">Draw with a finger or stylus — a signature or a photo is required.</p>
   315→    </div>
   316→  );
   317→}

   323→function PodDialog({ job, onClose, onCompleted }: { job: MyJob; onClose: () => void; onCompleted: () => void }) {
   324→  const [recipientName, setRecipientName] = useState(job.recipientName);
   325→  const [hasSignature, setHasSignature] = useState(false);
   326→  const [photoFile, setPhotoFile] = useState<File | null>(null);
   327→  const [photoUrl, setPhotoUrl] = useState<string | null>(null);
   328→  const [submitting, setSubmitting] = useState(false);
   329→  const [tried, setTried] = useState(false);
   330→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   331→  const photoUrlRef = useRef<string | null>(null);
   332→
   333→  useEffect(
   334→    () => () => {
   335→      if (photoUrlRef.current) URL.revokeObjectURL(photoUrlRef.current);
   336→    },
   337→    [],
   338→  );
   339→
   340→  const pickPhoto = (file: File | null) => {
   341→    if (photoUrlRef.current) {
   342→      URL.revokeObjectURL(photoUrlRef.current);
   343→      photoUrlRef.current = null;
   344→    }
   345→    setPhotoFile(file);
   346→    if (file) {
   347→      const url = URL.createObjectURL(file);
   348→      photoUrlRef.current = url;
   349→      setPhotoUrl(url);
   350→    } else {
   351→      setPhotoUrl(null);
   352→    }
   353→  };
   354→
   355→  const nameMissing = tried && !recipientName.trim();
   356→  const proofMissing = tried && !hasSignature && !photoFile;
   357→
   358→  const submit = async () => {
   359→    setTried(true);
   360→    const name = recipientName.trim();
   361→    if (!name) {
   362→      toast.error("Enter the recipient's name to continue");
   363→      return;
   364→    }
   365→    if (!hasSignature && !photoFile) {
   366→      toast.error("Capture a signature or attach a photo before completing");
   367→      return;
   368→    }
   369→    setSubmitting(true);
   370→    try {
   371→      const form = new FormData();
   372→      form.append("recipientName", name);
   373→      form.append("complete", "true");
   374→      if (hasSignature && canvasRef.current) {
   375→        const canvas = canvasRef.current;
   376→        const blob = await new Promise<Blob | null>((resolve) => canvas.toBlob(resolve, "image/png"));
   377→        if (!blob) throw new Error("Could not read the signature — try again or attach a photo");
   378→        form.append("file", new File([blob], `${job.trackingCode}-signature.png`, { type: "image/png" }));
   379→        form.append("kind", "SIGNATURE");
   380→      } else if (photoFile) {
   381→        form.append("file", photoFile, photoFile.name);
   382→        form.append("kind", "PHOTO");
   383→      } else {
   384→        throw new Error("No proof captured");
   385→      }
   386→      await api.post(`/api/deliveries/${job.id}/pod`, form);
   387→      toast.success("Delivery completed — proof recorded");
   388→      onCompleted();
   389→    } catch (e) {
   390→      toast.error(e instanceof Error ? e.message : "Could not record proof of delivery");
   391→    } finally {
   392→      setSubmitting(false);
   393→    }
   394→  };
   395→
   396→  return (
   397→    <Dialog open onOpenChange={(open) => { if (!open && !submitting) onClose(); }}>
   398→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-md">
   399→        <DialogHeader>
   400→          <DialogTitle>Complete delivery</DialogTitle>
   401→          <DialogDescription>
   402→            Record proof of delivery for <span className="font-mono font-medium text-foreground">{job.trackingCode}</span>.
   403→          </DialogDescription>
   404→        </DialogHeader>
   405→
   406→        <div className="flex flex-col gap-4">
   407→          <div className="flex flex-col gap-2">
   408→            <Label htmlFor="pod-recipient">Recipient name *</Label>
   409→            <Input
   410→              id="pod-recipient"
   411→              value={recipientName}
   412→              onChange={(e) => setRecipientName(e.target.value)}
   413→              placeholder="Who received the package?"
   414→              autoComplete="off"
   415→              className="h-11"
   416→              aria-invalid={nameMissing}
   417→            />
   418→            {nameMissing && <p className="text-xs text-rose-600 dark:text-rose-400">Recipient name is required.</p>}
   419→          </div>
   420→
   421→          <SignaturePad canvasRef={canvasRef} onInkChange={setHasSignature} />
   422→
   423→          <div className="flex items-center gap-3" aria-hidden="true">
   424→            <span className="h-px flex-1 bg-border" />
   425→            <span className="text-xs uppercase tracking-wide text-muted-foreground">or</span>
   426→            <span className="h-px flex-1 bg-border" />
   427→          </div>
   428→
   429→          <div className="flex flex-col gap-2">
   430→            <Label htmlFor="pod-photo">Photo proof (optional alternative)</Label>
   431→            <Input
   432→              id="pod-photo"
   433→              key={photoUrl ? "picked" : "empty"}
   434→              type="file"
   435→              accept="image/png,image/jpeg,image/webp"
   436→              onChange={(e) => pickPhoto(e.target.files?.[0] ?? null)}
   437→              className="h-11"
   438→            />
   439→            {proofMissing && (
   440→              <p className="text-xs text-rose-600 dark:text-rose-400">Add a signature or a photo to complete the delivery.</p>
   441→            )}
   442→            {photoUrl && photoFile ? (
   443→              <div className="flex items-start gap-3 rounded-lg border bg-muted/40 p-3">
   444→                {/* eslint-disable-next-line @next/next/no-img-element */}
   445→                <img src={photoUrl} alt="Photo proof preview" className="size-20 shrink-0 rounded-md border object-cover" />
   446→                <div className="flex min-w-0 flex-1 flex-col items-start gap-1">
   447→                  <p className="w-full truncate text-xs font-medium">{photoFile.name}</p>
   448→                  <p className="text-[11px] text-muted-foreground">{Math.max(1, Math.round(photoFile.size / 1024))} KB</p>
   449→                  <Button
   450→                    type="button"
   451→                    variant="outline"
   452→                    size="sm"
   453→                    className="mt-1 gap-1.5"
   454→                    onClick={() => pickPhoto(null)}
   455→                    disabled={submitting}
   456→                  >
   457→                    <X className="size-3.5" aria-hidden="true" /> Remove photo
   458→                  </Button>
   459→                </div>
   460→              </div>
   461→            ) : null}
   462→          </div>
   463→        </div>
   464→
   465→        <DialogFooter>
   466→          <Button type="button" variant="outline" onClick={onClose} disabled={submitting} className="min-h-11">
   467→            Cancel
   468→          </Button>
   469→          <Button type="button" onClick={submit} disabled={submitting} className="min-h-11">
   470→            {submitting ? <Loader2 className="size-4 animate-spin" aria-hidden="true" /> : <CheckCircle2 className="size-4" aria-hidden="true" />}
   471→            {submitting ? "Recording…" : "Complete delivery"}
   472→          </Button>
   473→        </DialogFooter>
   474→      </DialogContent>
   475→    </Dialog>
   476→  );
   477→}

   483→function ReportProblemDialog({ job, onClose, onReported }: { job: MyJob; onClose: () => void; onReported: () => void }) {
   484→  const [reason, setReason] = useState<string>("");
   485→  const [note, setNote] = useState("");
   486→  const [submitting, setSubmitting] = useState(false);
   487→  const [tried, setTried] = useState(false);
   488→
   489→  const submit = async () => {
   490→    setTried(true);
   491→    if (!reason) {
   492→      toast.error("Select what went wrong");
   493→      return;
   494→    }
   495→    setSubmitting(true);
   496→    try {
   497→      const body: { status: "FAILED"; failureReason: string; note?: string } = {
   498→        status: "FAILED",
   499→        failureReason: reason,
   500→      };
   501→      if (note.trim()) body.note = note.trim();
   502→      await api.post(`/api/deliveries/${job.id}/status`, body);
   503→      toast.success("Problem reported — the dispatcher has been notified");
   504→      onReported();
   505→    } catch (e) {
   506→      toast.error(e instanceof Error ? e.message : "Could not report the problem");
   507→    } finally {
   508→      setSubmitting(false);
   509→    }
   510→  };
   511→
   512→  return (
   513→    <Dialog open onOpenChange={(open) => { if (!open && !submitting) onClose(); }}>
   514→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-md">
   515→        <DialogHeader>
   516→          <DialogTitle>Report a problem</DialogTitle>
   517→          <DialogDescription>
   518→            Mark <span className="font-mono font-medium text-foreground">{job.trackingCode}</span> as failed and alert your dispatcher.
   519→          </DialogDescription>
   520→        </DialogHeader>
   521→
   522→        <div className="flex flex-col gap-4">
   523→          <div className="flex flex-col gap-2">
   524→            <Label>What went wrong? *</Label>
   525→            <Select value={reason} onValueChange={setReason}>
   526→              <SelectTrigger className="h-11 w-full" aria-label="Failure reason">
   527→                <SelectValue placeholder="Select a reason" />
   528→              </SelectTrigger>
   529→              <SelectContent>
   530→                {FAILURE_REASONS.map((r) => (
   531→                  <SelectItem key={r.code} value={r.code}>
   532→                    {r.label}
   533→                  </SelectItem>
   534→                ))}
   535→              </SelectContent>
   536→            </Select>
   537→            {tried && !reason && <p className="text-xs text-rose-600 dark:text-rose-400">Select a reason so dispatch knows what happened.</p>}
   538→          </div>
